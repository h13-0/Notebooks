# 2 拓扑排序

拓扑排序是针对有向无环图的一种线性排序方法，其：
- 用于解决有向无环图(DAG)中的依赖顺序问题。
- <span style="background:#fff88f"><font color="#c00000">可以用于检测图是否有环</font></span>

## 2.1 Kahn算法

其基本思路是：
0. 准备：
	1. 构造依赖图
	2. 构造入度数组，成员是每个节点的入度值
	3. 创建"<font color="#c00000">已删除的依赖</font>"<font color="#c00000">的计数器</font>
1. 找到所有入度为0的节点，并生成队列
2. 从队列中<font color="#c00000">取出</font><span style="background:#fff88f"><font color="#c00000">并删除元素</font></span>，并在图中<font color="#c00000">删除节点和依赖关系</font>
3. 删除上述依赖关系时，<font color="#c00000">要对配对点的入度</font> `--`
4. 若上述点入度归零，则加入队列
5. 重复直到队列为空

### 2.1.1 拓扑排序

```CPP
class Solution {
public:
    vector<int> getTaskOrder(
        int taskCount,
        const vector<vector<int>>& dependencies
    ) {
        vector<vector<int>> graph(taskCount);
        vector<int> indegree(taskCount, 0);

        for(const auto& dependency : dependencies) {
            int from = dependency[0];
            int to = dependency[1];

            graph[from].push_back(to);
            indegree[to]++;
        }

        deque<int> readyTasks;

        for(int task = 0; task < taskCount; task++) {
            if(indegree[task] == 0) {
                readyTasks.push_back(task);
            }
        }

        vector<int> order;
        order.reserve(taskCount);

        while(!readyTasks.empty()) {
            int task = readyTasks.front();
            readyTasks.pop_front();

            order.push_back(task);

            for(int nextTask : graph[task]) {
                if(--indegree[nextTask] == 0) {
                    readyTasks.push_back(nextTask);
                }
            }
        }

        // 没能处理全部任务，说明存在循环依赖
        if(order.size() != taskCount) {
            return {};
        }

        return order;
    }
};
```

```CPP
class Solution {
public:
    bool canCompleteAllTasks(int taskCount, const vector<vector<int>>& dependencies) {
        vector<vector<int>> graph(taskCount);
        vector<int> indegree(taskCount, 0);

        // dependency = {from, to}
        // 表示 from 完成后，to 才能执行
        for(const auto& dependency : dependencies) {
            int from = dependency[0];
            int to = dependency[1];

            graph[from].push_back(to);
            indegree[to]++;
        }

        deque<int> readyTasks;

        // 没有前置依赖的任务可以直接执行
        for(int task = 0; task < taskCount; task++) {
            if(indegree[task] == 0) {
                readyTasks.push_back(task);
            }
        }

        int completed = 0;

        while(!readyTasks.empty()) {
            int task = readyTasks.front();
            readyTasks.pop_front();

            completed++;

            // 当前任务完成后，释放它的后继任务
            for(int nextTask : graph[task]) {
                indegree[nextTask]--;

                if(indegree[nextTask] == 0) {
                    readyTasks.push_back(nextTask);
                }
            }
        }

        return completed == taskCount;
    }
};
```
