# 2 基本思路

其基本思路为：
1. 使用深度优先遍历，找到目标点
2. 在退栈的过程中，记录退栈路径
3. 退栈路径即为根路径

以[[数据结构/数据结构算法/LeetCode/LeetCode题目汇总#236 二叉树的最近公共祖先|LeetCode 236]]为例：
![[数据结构/数据结构算法/LeetCode/LeetCode题目汇总#236 二叉树的最近公共祖先]]

```CPP
class Solution {
public:
    // 找到目标节点的根路径
    bool findRoute(TreeNode* node, TreeNode* target, deque<TreeNode*>& route) {
        if(node == nullptr) {
            return false;
        }

        if(node == target) {
            route.push_back(node);
            return true;
        }

        if(
        	findRoute(node->left, target, route) || 
        	findRoute(node->right, target, route)
        ) {
            // 当左右子树中出现路径时，把当前节点加入路径前列
            route.push_front(node);
            return true;
        }

        return false;
    }

    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        deque<TreeNode*> pRoute;
        deque<TreeNode*> qRoute;

        findRoute(root, p, pRoute);
        findRoute(root, q, qRoute);

        TreeNode* answer = nullptr;

        for(int i = 0; i < min(pRoute.size(), qRoute.size()); i++) {
            if(pRoute[i] != qRoute[i]) {
                break;
            }
            answer = pRoute[i];
        }

        return answer;
    }
};
```
